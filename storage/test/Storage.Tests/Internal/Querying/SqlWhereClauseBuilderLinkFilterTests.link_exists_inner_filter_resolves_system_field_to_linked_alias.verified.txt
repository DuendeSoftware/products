EXISTS (
    SELECT 1 FROM main.entity_links lk0
    JOIN main.entities le0
      ON le0.pool_id = lk0.pool_id
      AND le0.entity_type_id = lk0.right_entity_type_id
      AND le0.entity_id = lk0.right_entity_id
    WHERE lk0.pool_id = v.pool_id
      AND lk0.link_type_id = @link_type_0
      AND lk0.left_entity_type_id = v.entity_type_id
      AND lk0.left_entity_id = v.entity_id
      AND (le0.created_at = @value_1)
)

Parameters:
@link_type_0 = 9912
@value_1 = 2024-01-15T00:00:00.0000000Z
